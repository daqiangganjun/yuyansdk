cmake_minimum_required(VERSION 3.22)
project(selfopt_rime LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(BUILD_TEST OFF CACHE BOOL "" FORCE)
set(BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(BUILD_STATIC ON CACHE BOOL "" FORCE)
set(ENABLE_TIMESTAMP OFF CACHE BOOL "" FORCE)
set(DEPS_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/.cache" CACHE PATH "Pinned native sources")

set(BOOST_INCLUDE_LIBRARIES algorithm crc dll interprocess range regex scope_exit signals2 utility uuid)
add_subdirectory("${DEPS_ROOT}/boost" boost EXCLUDE_FROM_ALL)
file(GLOB boost_headers "${DEPS_ROOT}/boost/libs/*/include" "${DEPS_ROOT}/boost/libs/*/*/include")
include_directories(${boost_headers})
add_compile_definitions(BOOST_DLL_USE_STD_FS)

set(WITH_GFLAGS OFF CACHE BOOL "" FORCE)
set(WITH_GTEST OFF CACHE BOOL "" FORCE)
set(WITH_UNWIND OFF CACHE BOOL "" FORCE)
add_subdirectory("${DEPS_ROOT}/librime/deps/glog" glog)
set(Glog_INCLUDE_PATH "${DEPS_ROOT}/librime/deps/glog/src" CACHE PATH "" FORCE)
set(Glog_LIBRARY glog CACHE STRING "" FORCE)

set(YAML_CPP_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(YAML_CPP_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(YAML_CPP_INSTALL OFF CACHE BOOL "" FORCE)
add_subdirectory("${DEPS_ROOT}/librime/deps/yaml-cpp" yaml-cpp)
set(YamlCpp_INCLUDE_PATH "${DEPS_ROOT}/librime/deps/yaml-cpp/include" CACHE PATH "" FORCE)
set(YamlCpp_LIBRARY yaml-cpp CACHE STRING "" FORCE)
set(YamlCpp_NEW_API TRUE CACHE BOOL "" FORCE)

set(SNAPPY_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(SNAPPY_BUILD_BENCHMARKS OFF CACHE BOOL "" FORCE)
set(SNAPPY_INSTALL OFF CACHE BOOL "" FORCE)
add_subdirectory("${DEPS_ROOT}/snappy" snappy)
set(HAVE_SNAPPY TRUE)
set(LEVELDB_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LEVELDB_BUILD_BENCHMARKS OFF CACHE BOOL "" FORCE)
set(LEVELDB_INSTALL OFF CACHE BOOL "" FORCE)
add_subdirectory("${DEPS_ROOT}/librime/deps/leveldb" leveldb)
set(LevelDb_INCLUDE_PATH "${DEPS_ROOT}/librime/deps/leveldb/include" CACHE PATH "" FORCE)
set(LevelDb_LIBRARY leveldb CACHE STRING "" FORCE)

set(ENABLE_TOOLS OFF CACHE BOOL "" FORCE)
add_subdirectory("${DEPS_ROOT}/librime/deps/marisa-trie" marisa)
set(Marisa_INCLUDE_PATH "${DEPS_ROOT}/librime/deps/marisa-trie/include" CACHE PATH "" FORCE)
set(Marisa_LIBRARY marisa CACHE STRING "" FORCE)
set(LIBMARISA marisa)
set(USE_SYSTEM_MARISA ON CACHE BOOL "" FORCE)
add_subdirectory("${DEPS_ROOT}/librime/deps/opencc" opencc)
set(Opencc_INCLUDE_PATH "${DEPS_ROOT}/librime/deps/opencc/src" CACHE PATH "" FORCE)
set(Opencc_LIBRARY libopencc CACHE STRING "" FORCE)
file(GLOB opencc_headers "${Opencc_INCLUDE_PATH}/*.hpp" "${CMAKE_BINARY_DIR}/opencc/src/opencc_config.h")
file(COPY ${opencc_headers} DESTINATION "${CMAKE_BINARY_DIR}/include/opencc")
include_directories("${CMAKE_BINARY_DIR}/include" "${DEPS_ROOT}/librime/include")
add_subdirectory("${DEPS_ROOT}/librime" librime)

add_library(selfopt_rime SHARED rime_jni.cpp)
target_include_directories(selfopt_rime PRIVATE "${DEPS_ROOT}/librime/src"
    "${DEPS_ROOT}/librime/plugins/librime-predict/src" "${CMAKE_BINARY_DIR}/librime/src"
    "${Glog_INCLUDE_PATH}" "${CMAKE_BINARY_DIR}/glog" "${YamlCpp_INCLUDE_PATH}")
target_compile_definitions(selfopt_rime PRIVATE GLOG_USE_GLOG_EXPORT YAML_CPP_STATIC_DEFINE)
target_compile_options(selfopt_rime PRIVATE -Wno-deprecated-declarations)
target_link_libraries(selfopt_rime PRIVATE -Wl,--whole-archive rime-static -Wl,--no-whole-archive log)
target_link_options(selfopt_rime PRIVATE -Wl,-z,max-page-size=16384 -Wl,--exclude-libs,ALL)
